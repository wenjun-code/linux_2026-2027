Выполнение работы

1. getopts и grep

while getopts "i:o:p:Cnh" opt; do
  case $opt in
    i) IN="$OPTARG" ;;
    o) OUT="$OPTARG" ;;
    p) PAT="$OPTARG" ;;
    C) SENS=1 ;;
    n) NUMS=1 ;;
  esac
done
grep $OPTS -- "$PAT" "$IN"
  • -C — учитывать регистр (без ключа — grep -i)
  • -n — номера строк, -o — вывод в файл
  • -i, -p — обязательные ключи

2. Программа на C и код возврата

if (n > 0) return 0;   /* больше нуля */
if (n < 0) return 1;   /* меньше нуля */
return 3;              /* равно нулю  */
echo "$x" | ./sign
case $? in
  0) echo "больше нуля" ;;
  1) echo "меньше нуля" ;;
  3) echo "равно нулю" ;;
esac
  • коды возврата: 0 / 1 / 3
  • командный файл анализирует $?

sign.c, checknum.sh и анализ $?

3. Создание N файлов

MODE=create
[ "$1" = "-d" ] && { MODE=delete; shift; }
n=1
while [ "$n" -le "$N" ]; do
  if [ "$MODE" = create ]; then touch "$n.tmp"
  else rm -f "$n.tmp"; fi
  n=$((n+1))
done
  • ./mkfiles.sh 5 — создать 1.tmp…5.tmp
  • ./mkfiles.sh -d 5 — удалить их
  • цикл while с инкрементом

4. Упаковка в tar

find "$DIR" -maxdepth 1 -type f -mtime -7 -print0 \
  | tar -czf "$ARCH" --null -T -
  • -mtime -7 — изменённые менее недели назад
  • -print0 + tar --null -T - — безопасно для имён с пробелами
  • tar -tzf — проверка содержимого архива
